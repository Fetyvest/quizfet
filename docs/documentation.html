config.json

"game"

"game" (obj) — глобальные настройки игровой сессии
  • "title" (str) — название квиза
  • "subtitle" (null|str) — подзаголовок квиза
    • null — подзаголовок отсутствует
    • str — текст подзаголовка
  • "showAnswers" (bool)
    • true — показывать правильный ответ текущего вопроса перед следующим
    • false — сразу переходить к следующему вопросу
  • "waitAdmin" (bool)
    • true — админ сам запускает каждый следующий вопрос
    • false — каждый игрок решает задания в своём темпе
  • "showQuestionCount" (bool)
    • true — показывать общее количество вопросов в квизе
    • false — показывать количество только уже решённых вопросов
  • "suffleQuestions" (bool)
    • true — перемешивать порядок вопросов при старте игры
    • false — идти строго по заданному порядку
  • "suffleAnswers" (bool)
    • true — перемешивать варианты ответов внутри вопросов
    • false — отображать варианты в исходном порядке
  • "playerLimit" (null|int) — максимальное количество игроков в игровой комнате
    • null — без ограничений
    • int — количество игроков
  • "playerLimitQuestion" (obj) — максимальное количество игроков на одном вопросе
    • "min" (int) — минимальное количество игроков
    • "max" (null|int) — максимальное количество игроков
      • null — без ограничений
      • int — количество игроков
  • "preview" (bool{false}|obj)
    • false — полностью выключает экран предпросмотра
    • obj — настройки отображения экрана предпросмотра
      • "time" (float) — длительность показа экрана предпросмотра в секундах
      • "showQuestion" (bool)
        • true — показывать формулировку вопроса во время предпросмотра
        • false — скрывать текст вопроса во время предпросмотра
      • "showAnswers" (bool)
        • true — заранее показывать варианты ответа на экране предпросмотра
        • false — держать варианты скрытыми до начала таймера ответа
  • "score" (bool{false}|obj) — настройки калькуляции и отображения очков
    • false — выключить систему очков
    • obj — настройки счёта
      • "isShow" (obj) — управление видимостью очков
        • "my" (bool)
          • true — игрок видит своё количество очков
          • false — свой счёт скрыт
        • "others" (bool{false}|string{nearby|all})
          • false — отображение количества очков других игроков скрыто
          • "nearby" — отображение количества очков соседних игроков в таблице
          • "all" — отображение счёта всех игроков
      • "magnitude" (int) — базовая величина количества очков за правильный ответ. Например, 0 - до 10, 1 - до 100, 10 - до 1000
      • "multiply" (float) — глобальный множитель набранных очков
      • "limitAnswerMultiplier" (obj) — границы количества очков за один вопрос
        • "min" (float) — минимальное количество очков (результат равен параметру magnitude, умноженному на данный параметр)
        • "max" (float|null) — максимальное количество очков (результат равен параметру magnitude, умноженному на данный параметр)
          • float — числовой множитель
          • null — без ограничений
      • "limitGameMultiplier" (obj) — границы количества очков за всю игру
        • "min" (float) — минимальное количество очков (результат равен параметру magnitude, умноженному на данный параметр)
        • "max" (float|null) — максимальное количество очков (результат равен параметру magnitude, умноженному на данный параметр)
          • float — числовой множитель
          • null — без ограничений
      • "timeDependence" (bool{false}|obj) — динамика уменьшения очков в зависимости от времени размышления
        • false — выключить зависимость от времени
        • "time" (obj) — временные рамки фиксации скорости
          • "min" (float) — порог минимального времени, быстрее которого очки перестают увеличиваться
          • "max" (float|null) — время в секундах, до истечения которого начисляется максимум очков
            • float — числовой множитель
            • null — без ограничений
          • "minPercent" (int) — минимальный процент очков от базовой стоимости, ниже которого награда не опустится даже при ответе на последней секунде
      • "rankDependenceMultiplier" (float) — каждый следующий игрок в топе получит количество очков, умноженное на это значение
      • "lastRankBonusMultiplier" (float) — последний игрок в топе будет получать количество очков, умноженное на это значение
      • "streakBonus" (boll{false}|obj) — логика бонусов за серию правильных ответов подряд
        • false — выключить бонус за серию
        • obj — настройки бонусов за серии
          • "percent" (float) — процент бонуса, добавляемый за каждый шаг серии
          • "type" (string) — база для расчета процента
            • "total" — процент рассчитывается от общего накопленного счёта игрока
            • "answer" — процент считается только от стоимости текущего ответа
          • "includeSaveStreak" (bool) — учитывается ли накопленный стрик, защищенный сохранением, при расчете этого бонуса
      • "substraction" (bool{false}|obj) — параметры штрафов при совершении ошибок
        • false — выключить вычитание очков
        • obj — настройки штрафов
          • "percent" (int) — процент, удерживаемый в качестве штрафа
          • "type" (string) — объект удержания штрафа
            • "total" — штраф списывается напрямую с общего баланса очков игрока
            • "answer" — штраф вычитается из потенциальной стоимости текущего вопроса (уход в минус)
  • "streak" (bool{false}|obj) — параметры системы серий ответов (страйков)
    • false — выключить серии
    • obj — настройки серий
      • "saveStreakCount" (int) — количество доступных «сохранений» (щитов), защищающих серию от сброса при ошибке
      • "includeSaveStreak" (bool) — засчитывать ли защищённый страйк в общее число
  • "rank" (obj) — архитектура таблицы лидеров
    • "isShow" (bool{false}|obj) — правила отображения рейтинга
      • false — выключить отображение рейтинга
      • obj — настройки отображения рейтинга
        • "my" (bool) — видит ли пользователь свое место в топе
        • "others" (bool{false}|string{nearby|all}) — режим просмотра чужих позиций
          • false — отображение мест других игроков скрыто
          • nearby — отображение места соседних игроков в таблице
          • all — отображение места всех игроков
    • "regarding" (string) — основной критерий сортировки ранка
      • "correct" — сортировка по количеству правильных ответов
      • "score" — сортировка по сумме набранных очков
  • "time" (obj) — глобальные лимиты времени сессии
    • "answer" (obj) — лимиты на размышление над одним вопросом
      • "min" (float) — минимально допустимый лимит времени (в секундах)
      • "max" (null|float) — максимально допустимый лимит времени (в секундах)
        • null — без ограничений
        • float — время в секундах
    • "game" (obj) — тайминги на прохождение всего квиза целиком
      • "min" (float) — минимальное общее время на игру (в секундах)
      • "max" (null|float) — максимальное общее время на игру (в секундах)
        • null — без ограничений
        • float — время в секундах
      • "lastChance" (bool) — если общее время квиза истекло, но игрок прямо сейчас решает вопрос
        • true — дать возможность ответить на текущий вопрос
        • false — мгновенно прервать игру

"questions"

"questions" (array) — массив с объектами вопросов. Каждый вопрос может содержать следующие параметры:
  • "type" (string{selection|input|order|match|container}) — механика взаимодействия с вопросом
    • "selection" — тестовый выбор из предложенных вариантов ответов (один или несколько правильных вариантов)
    • "input" — ручной текстовый ввод ответа в поле
    • "order" — сортировка вариантов в правильной последовательности
    • "match" — сопоставление пар между двумя группами элементов
    • "container" — распределение вариантов по соответствующим категориям
  • "title" (string) — основной текст (формулировка) вопроса. Поддерживает UTF-8 кодировку
  • "subtitle" (string) — подзаголовок, дополнительное описание, подсказка или контекст к вопросу
  • "image" (string) — ссылка на изображение к вопросу (может быть абсолютным URL "https://..." или локальным путем "./img.png")
  • "hint" (obj) — настройки подсказки для текущего вопроса
    • "time" (int) — задержка в секундах от начала вопроса, после которой подсказка станет доступна
    • "auto" (bool) — показывать ли подсказку автоматически при наступлении времени таймера без клика пользователя
    • "penalty" (bool|obj) — штраф за использование подсказки
      • false — штраф отсутствует
      • obj — параметры удержания штрафа
        • "substractionPercent" (float) — какой процент стоимости вопроса отнять при открытии подсказки
        • "ifWrong" (bool) — применять ли этот штраф даже в случае неверного ответа
    • "text" (string) — содержательный текст подсказки
  • "game" (obj) — локальное переопределение настроек. Позволяет точечно переписать глобальные параметры исключительно для данного вопроса. Не все параметры подлежат переопределению
  • "answers" (obj) — блок валидации и вариантов ответов. Структура динамически зависит от параметра "type" вопроса:
    [Для type: "selection"]
    • "variants" (array) — плоский массив с вариантами ответов (строками или числами)
    • "correct" (int|array) — указатель на верный выбор.
      • int — индекс одного правильного ответа
      • array — индексы множественного выбор
    • "isPatrialScore" (bool{false}|obj) — логика частичных баллов за распределение предметов по контейнерам
      • false — частичное количество баллов не выдаётся
      • obj — параметры частичных баллов
        • "emptyIsWrong" (bool)
          • true — считать невыбранный правильный ответ за ошибку
          • false — игнорировать невыбранные правильные ответы
    • "explanation" (obj|array) — пояснения к ответам
      • obj — делит подсказки на категории (правильный и неправильный ответ)
        • "correct" (string) — выводится при верном ответе
        • "wrong" (string) — выводится при ошибке
      • array — подсказки индивидуальны для каждого ответа из элемента "variants" по соответствующему индексу
    [Для type: "input"]
    • "correct" (string|array|null)
      • string — единственный правильный ответ
      • array — список всех допустимых строк-синонимов, которые система примет как правильный ответ
      • null — админ сам проверит ответ вопрос
    • "allowSymbolGroup" (array) — белый список разрешенных групп символов для ввода. Список строк, которые может содержать массив:
      • "numbers" — цифры разрешены
      • "english" — английские буквы разрешены
      • "space" — пробел разрешён разрешены
      • "math" — знаки математических операций разрешены
      • "specifical" — специальные символы разрешены
      • "other" — всё остальное разрешено
    • "isCorrectCase" (bool)
      • true — регистр букв должен строго совпадать с регистром букв ответа
      • false — регистр игнорируется (всегда, если язык не русский и не английский)
    • "showCountSymbols" (bool)
      • true — показывать количество символов в правильном ответе
      • false — количество символов правильного ответа скрыто
    • "trimAnswer" (bool)
      • true — автоматически отсекать лишние пробелы в начале и конце введенной пользователем строки перед проверкой
      • false — не отсекать лишние пробелы и оставить исходный ответ
    • "allowMultiString" (bool)
      • true — разрешить ввод многострочного текста
      • false — однострочный ответ
    • "done" (obj) — параметры форматирования строки вывода на финальном экране/экране истории
      • "before" (string) — текст, который отображается пользователю перед вводимым ответом
      • "after" (string) — текст, который отображается пользователю после вводимого ответом
    [Для type: "order"]
    • "variants" (array) — элементы, которые игроку необходимо расставить в правильном порядке
    • "correct" (array) — массив, задающий правильную последовательность индексов исходного массива вариантов
    • "isPatrialScore" (bool{false}|obj) — настройки частичного начисления баллов при частичной верной сортировке
      • false — частичное количество баллов не выдаётся
      • obj — параметры частичных баллов
          • "pairWrong" (bool)
            • true — если перепутана только 1 пара элементов (строго 2 элемента из всей последовательности неверны), то это считается за одну ошибку
            • false — если перепутана пара элементов, то это считается за 2 ошибки (так как два элемента не на своих местах)
          • "method" (string{index|accuracy|skip|islands|biggestIsland})
            • "index" — сравнивает по индексам. Из ряда 14325678 находит ошибочными 432 (3 ошибки)
            • "accuracy" — следующий пункт должен быть больше текущего. Из ряда 12635748 находит ошибочными 354 (3 ошибки)
            • "skip" — следующий пункт должен быть на 1 больше текущего. Из ряда 12678345 находит ошибочными 678 (3 ошибки)
            • "islands" — ищет цельные кусочки, не важен их порядок, но важен порядок элементов внутри. Из ряда 45612378 находит 3 ошибки, так как 3 кусочка
            • "biggestIsland" — ищет наибольший цельный кусочек, не важен их порядок, но важен порядок элементов внутри. Из ряда 78345612 находит ошибочными 7812 (4 ошибки)
          • "emptyIsWrong" (bool)
            • true — считать невыбранный правильный ответ за ошибку
            • false — игнорировать невыбранные правильные ответы

    [Для type: "match"]
    • "variants" (array) — две сопоставляемые группы данных. Массив содержит два массива (две группы элементов)
    • "correct" (array) — массив пар соответствий. Каждая пара представлена индексом элемента из первой группы и индексом из второй группы
    • "isPatrialScore" (obj) — настройки начисления баллов за неполное сопоставление
      • false — частичное количество баллов не выдаётся
      • obj — параметры частичных баллов
        • "pairWrong" (bool)
          • true — если перепутана только 1 пара элементов (строго 2 элемента из одной группы), то это считается за одну ошибку
          • false — если перепутана пара элементов, то это считается за 2 ошибки (так как два элемента не со своей парой)
        • "emptyIsWrong" (bool)
          • true — считать невыбранный правильный ответ за ошибку
          • false — игнорировать невыбранные правильные ответы

    [Для type: "container"]
    • "variants" (array) — структура распределения, содержит массив имен категорий (контейнеров) и массив перетаскиваемых элементов
    • "correct" (array) — привязка элементов к категориям (двумерный массив), порядок массивов внутри соответствует порядку категорий, внутри массивов индексы перетаскиваемых элементов
    • "isPatrialScore" (obj) — логика частичных баллов за распределение предметов по контейнерам
      • false — частичное количество баллов не выдаётся
      • obj — параметры частичных баллов
        • "emptyIsWrong" (bool)
          • true — считать невыбранный правильный ответ за ошибку
          • false — игнорировать невыбранные правильные ответы
